🚀 快速上手

from gaskill import Complex, Frac, vec2
from gaskill.tester import TestCase, test, TestRunner

# 复数 — 支持字符串解析
c = Complex("3+4j")
print(c * c) # (-7+24i)

# 分数 — 精确有理数
f = Frac(1, 3) + Frac(1, 6)
print(f) # 1/2

# 2D 向量 — 几何运算
v = vec2(3, 4)
print(v.length()) # 5.0

# 测试框架 — 零依赖 unittest 风格
class MyTest(TestCase):
    @test
    def test_add(self):
        self.assert_equal(1 + 1, 2)

TestRunner().run_all()
